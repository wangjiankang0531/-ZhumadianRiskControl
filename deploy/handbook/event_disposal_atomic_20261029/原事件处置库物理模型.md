# 原事件处置库物理模型

本文件由 `Application/tools/generate_event_disposal_database_docs.py` 从当前本机 MySQL 的两份只读 `SHOW CREATE TABLE` 结果生成。该结构已经在真实 MySQL 校验，不能据此声称全部业务 HTTP、角色或 Apache 环境已验收。

事务修正前后两表全行与 DDL 摘要相同；没有字段、索引、状态枚举或归属定义增量。目标状态仍是原 `research_status=2`，失败时同事务撤销状态与新处置记录。

## event_info

原实例结构 SHA256：`0f9b55341a17685a8e905eed3c427709e07b61c56ceacc324848db01d09f036d`。

```sql
CREATE TABLE `event_info` (
  `id` varchar(32) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NOT NULL COMMENT '主键ID',
  `point_id` varchar(32) CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL COMMENT '巡检点id',
  `task_id` varchar(32) CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL COMMENT '下发任务id',
  `event_number` varchar(32) CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL COMMENT '事件编号',
  `event_name` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL COMMENT '事件名称',
  `event_location` varchar(32) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '事件区域',
  `event_info` varchar(32) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '事件区域',
  `event_content` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL COMMENT '事件内容',
  `event_type` varchar(16) CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT '0' COMMENT '事件类型',
  `report_time` timestamp NULL DEFAULT NULL COMMENT '上报时间',
  `reporter_name` varchar(32) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '上报人',
  `event_level` varchar(16) CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL COMMENT '事件类别:0-特级；1-一般',
  `report_type` varchar(5) CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL COMMENT '上报类型:0-初报；1-信息记录',
  `event_ids` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL COMMENT '重复事件ids(当上报类型为0时可能存在多个重复事件id；当上报类型为1时可以存放初报事件id)',
  `event_status` varchar(5) CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL COMMENT '事件状态:0-预警期；1-发生期；2-控制期；3-恢复期',
  `research_status` varchar(5) CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT '0' COMMENT '研判状态:0-待研判；1-已研判',
  `research_result` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL COMMENT '研判结果',
  `is_task_primary` varchar(5) CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL COMMENT '是否为主关联任务信息:0-否；1-是',
  `warn_start` varchar(8) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '预警启动',
  `emergency_plan_id` varchar(36) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '应急预案id',
  `file_url` text CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci COMMENT '附件',
  `coordinate` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '坐标',
  `create_dept` varchar(32) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '创建部门',
  `deleted` char(1) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT '0' COMMENT '是否已删除',
  `created_by` varchar(50) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '创建人',
  `created_at` timestamp NULL DEFAULT NULL COMMENT '创建时间',
  `updated_by` varchar(50) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '修改人',
  `updated_at` timestamp NULL DEFAULT NULL COMMENT '修改时间',
  `flow_id` varchar(50) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '流程id',
  PRIMARY KEY (`id`) USING BTREE,
  KEY `idx_report_time` (`report_time`) USING BTREE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='上报事件表'
```

## event_disposal_record

原实例结构 SHA256：`8c74b0152cd4c735a718e858ab139077e817421cb6f44db3a407b041f5249dca`。

```sql
CREATE TABLE `event_disposal_record` (
  `id` int NOT NULL AUTO_INCREMENT COMMENT '自增主键',
  `event_id` varchar(32) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NOT NULL COMMENT '关联的事件ID，用于绑定具体事件',
  `description` text CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NOT NULL COMMENT '处置记录描述',
  `attachments` text CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci COMMENT '附件路径，多个附件用逗号分隔（例如：/file1.pdf,/file2.jpg）',
  `username` varchar(64) DEFAULT NULL,
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '记录创建时间',
  PRIMARY KEY (`id`) USING BTREE,
  KEY `idx_event_id` (`event_id`) USING BTREE COMMENT '为事件ID创建索引，提升关联查询效率'
) ENGINE=InnoDB AUTO_INCREMENT=45 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='事件处置记录表（用于记录事件的时间处置过程）'
```
