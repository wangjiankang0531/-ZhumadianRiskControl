# 原应急演练与附件库物理模型

由 `Application/tools/generate_emergency_exercise_database_docs.py` 从实际本机 MySQL 的两份只读 `SHOW CREATE TABLE` 自动生成。本文件对应原南港 MySQL，不对应项目早期 mock SQLite 模型。

当前为离线事务候选，未安装、未重启、未发布公网。没有字段、索引、状态枚举、附件归属定义变更。真实 MySQL 临时表测试已经通过；实际应用装配、正常用户 HTTP 与角色验收仍待。

## emergency_exercise

实际结构 SHA256：`906bbb8dfa9f2c8c75bb469355a18e84b220fe56bfc4d92003ec1fba56428cc9`。

```sql
CREATE TABLE `emergency_exercise` (
  `id` varchar(32) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NOT NULL COMMENT '应急预案单位主键',
  `emergency_plan_id` varchar(32) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '应急预案主键',
  `exercise_location` varchar(100) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '演练地点',
  `organize_department` varchar(50) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '组织部门',
  `commander_chief` varchar(50) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '总指挥',
  `exercise_time` timestamp NULL DEFAULT NULL COMMENT '演练时间',
  `participants` varchar(200) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '参加部门及人员',
  `cooperating_unit` varchar(200) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '配合单位',
  `exercise_type` varchar(5) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '演练类别',
  `exercise_device` varchar(200) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '演练设备及器材',
  `exercise_describe` varchar(500) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '演练内容及过程描述',
  `exercise_special_situation` varchar(500) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '演练过程中特殊处理情况',
  `created_at` timestamp NULL DEFAULT NULL COMMENT '创建时间',
  `problems_andmeasures` varchar(500) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '问题及改进措施',
  PRIMARY KEY (`id`) USING BTREE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='应急预案-演练记录'
```
