cmake_minimum_required(VERSION 3.20)
project(XnbyPageLifecycleProbe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(HT_PLATFORM_WINDOWS ON)
find_package(fmt CONFIG REQUIRED)
find_package(spdlog CONFIG REQUIRED)
find_package(glm CONFIG REQUIRED)
find_package(Vulkan REQUIRED)
find_package(volk CONFIG REQUIRED)
find_package(VulkanMemoryAllocator CONFIG REQUIRED)
find_package(SDL2 CONFIG REQUIRED)
add_subdirectory(XNBYEngine/core)
add_subdirectory(XNBYEngine/rhi)
add_executable(native_page_rhi_probe native_page_rhi_probe.cpp)
target_include_directories(native_page_rhi_probe PRIVATE XNBYEngine/renderer/include XNBYEngine/asset/include ${CMAKE_PREFIX_PATH}/include)
target_link_libraries(native_page_rhi_probe PRIVATE ht_rhi SDL2::SDL2)
target_compile_options(native_page_rhi_probe PRIVATE /W4 /utf-8)
