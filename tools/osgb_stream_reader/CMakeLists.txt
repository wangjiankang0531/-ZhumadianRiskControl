cmake_minimum_required(VERSION 3.25)
project(ZhumadianReadOnlyOsgb LANGUAGES CXX)
add_executable(zmd_osgb_stream_reader main.cpp)
target_compile_features(zmd_osgb_stream_reader PRIVATE cxx_std_17)
target_include_directories(zmd_osgb_stream_reader PRIVATE "${ZMD_OSG_SOURCE}/include" "${ZMD_OSG_BUILD}/include" "${ZMD_JSON_INCLUDE}")
target_link_libraries(zmd_osgb_stream_reader PRIVATE "${ZMD_OSG_BUILD}/lib/osg.lib" "${ZMD_OSG_BUILD}/lib/osgDB.lib" "${ZMD_OSG_BUILD}/lib/OpenThreads.lib")
if(MSVC)
 target_compile_options(zmd_osgb_stream_reader PRIVATE /utf-8 /W4)
 target_compile_definitions(zmd_osgb_stream_reader PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
